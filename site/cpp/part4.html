9.3 Performance habits that move the needle

// 1 · Compile-level (judge already uses -O2; know why it matters)
//    -O2 typically gives 5–20× over -O0. Inline small functions, mark hot lambdas
//    with [](...) rather than std::function.

// 2 · IO: the single biggest win for input-heavy problems
ios::sync_with_stdio(false);     // stop syncing C++ streams with C stdio
cin.tie(nullptr);                // stop flushing cout before every cin

// 3 · Reserve and avoid copies
vector<int> res; res.reserve(n);
for (const auto& x : big) { }     // const&: no copy per element
auto byValue = bigVector;          // ✗ O(n) copy
const auto& ref = bigVector;      // ✓

// 4 · Choose the right container (the algorithmic win)
//    membership: unordered_set O(1)  vs  vector O(n)
//    ordered queries: set/map O(log n)
//    top-K: priority_queue O(n log k) vs sort O(n log n)
//    prefix sums: O(1) range queries

// 5 · Cache friendliness (why vector beats list by 10× on real hardware)
//    contiguous memory → prefetcher works; pointer chasing → cache misses

// 6 · Avoid needless work in loops
int n = (int)v.size();                       // hoist
for (int i = 0; i < n; ++i) { ... }          // not v.size() each time
// compute prefix sums once instead of summing inside a loop (O(n²) → O(n))

// 7 · Recursion overhead: iterative DFS/BFS over recursion when depth is large
// 8 · Bitset tricks for set operations on 64/1024-bit sets (ch. 10)
// 9 · Move semantics: std::move on containers you will not use again
res.push_back(std::move(temp));              // avoid copying the buffer
vector<int> steal = std::move(bigVec);       // bigVec becomes empty

// 10 · Measure, do not guess
//   chrono::steady_clock around suspect blocks; count operations; try small→large input sizes.
Practical time budget for C++ on LeetCode (there is no fixed instruction count, but these are the orders of magnitude that decide pass/TLE with a 1–2 s limit): 10⁸ simple operations ≈ 0.2–0.5 s · 10⁷ unordered_map ops ≈ 0.5–1 s · 10⁷ map/set ops ≈ 2–5 s (too slow) · recursion on 10⁵ nodes is fine if depth stays O(log n) · O(n²) with n = 10⁴ is 10⁸ → borderline; with n = 10⁵ it never passes.

10 Bit manipulation and number theory

The tools for subset-enumeration, XOR puzzles, modular arithmetic and “O(1) math” questions.

10.1 Bit operations

// ---- the classic XOR problems -------------------------------------------
int singleNumber(const vector<int>& a) {           // every element twice, one once
    int x = 0; for (int v : a) x ^= v; return x;    // O(1) space, no hashing
}
pair<int,int> twoSingles(const vector<int>& a) {    // two elements appear once
    int xr = 0; for (int v : a) xr ^= v;
    int low = xr & -xr;                             // isolate a differing bit
    int g1 = 0, g2 = 0;
    for (int v : a) (v & low ? g1 : g2) ^= v;
    return {g1, g2};
}

// ---- counts and powers ---------------------------------------------------
__builtin_popcount(255);                            // 8
__builtin_popcountll(1ULL << 40);                   // 1
bool isPow2(long long x) { return x > 0 && (x & (x - 1)) == 0; }
int lowbit(int x) { return x & -x; }               // value of the lowest set bit

// ---- subset DP (bitmask over n ≤ 20 elements) ----------------------------
// dp[mask] = best value using exactly the elements in mask
int n = (int)cost.size();                           // problem input
vector<long long> dp(1 << n, 0);
for (int mask = 1; mask < (1 << n); ++mask) {
    int i = __builtin_ctz(mask);                    // lowest set index
    dp[mask] = dp[mask ^ (1 << i)] + cost[i];       // adapt this recurrence per problem
}
// Enumerate all submasks of a mask (total work is 3^n)
void enumerateSubmasks(int mask) {
    for (int sub = mask; sub; sub = (sub - 1) & mask) { /* use sub */ }
}

10.2 Number theory essentials

// ---- gcd / lcm -----------------------------------------------------------
long long gcdll(long long a, long long b) { return b ? gcdll(b, a % b) : a; }
long long lcmll(long long a, long long b) { return a / gcdll(a, b) * b; }   // divide FIRST

// ---- sieve of Eratosthenes: all primes up to n in O(n log log n) ----------
vector<int> primes;
vector<bool> isComposite(n + 1, false);
for (long long i = 2; i <= n; ++i) {
    if (!isComposite[i]) {
        primes.push_back((int)i);
        for (long long j = i * i; j <= n; j += i) isComposite[j] = true;    // i*i can overflow int!
    }
}
// O(n) linear sieve variant tracks the smallest prime factor (SPF) for factorisation:
vector<int> spf(n + 1);
for (int i = 2; i <= n; ++i) {
    if (!spf[i]) { spf[i] = i; primes.push_back(i); }
    for (int p : primes) { if (p > spf[i] || 1LL * i * p > n) break; spf[i * p] = p; }
}
// factorise x using spf
vector<pair<int,int>> factorise(int x, const vector<int>& spf) {
    vector<pair<int,int>> f;
    while (x > 1) { int p = spf[x], c = 0; while (x % p == 0) { x /= p; ++c; } f.push_back({p, c}); }
    return f;
}

// ---- modular arithmetic -------------------------------------------------
const long long MOD = 1'000'000'007;                 // prime
long long add(long long a, long long b) { return (a + b) % MOD; }
long long mul(long long a, long long b) { return a * b % MOD; }         // a,b < 1e9 → fits
long long mpow(long long b, long long e) { long long r = 1; b %= MOD;
    while (e) { if (e & 1) r = r * b % MOD; b = b * b % MOD; e >>= 1; } return r; }
long long inv(long long a) { return mpow(a, MOD - 2); }                 // Fermat
long long sub(long long a, long long b) { return ((a - b) % MOD + MOD) % MOD; }

// ---- combinatorics ------------------------------------------------------
// factorials + inverse factorials for O(1) nCr mod p (p prime)
vector<long long> fact(N + 1), invFact(N + 1);
fact[0] = 1;
for (int i = 1; i <= N; ++i) fact[i] = fact[i-1] * i % MOD;
invFact[N] = inv(fact[N]);
for (int i = N; i > 0; --i) invFact[i-1] = invFact[i] * i % MOD;
auto C = [&](int n, int k) {
    if (k < 0 || k > n) return 0LL;
    return fact[n] * invFact[k] % MOD * invFact[n-k] % MOD;
};

// ---- geometry / precision -------------------------------------------------
struct Pt { long long x, y; };
long long cross(const Pt& a, const Pt& b, const Pt& c) {   // >0: left turn (ccw)
    return (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x);
}
bool onSegment(const Pt& a, const Pt& b, const Pt& p) {  // collinear + within bbox
    return cross(a, b, p) == 0 && min(a.x,b.x) <= p.x && p.x <= max(a.x,b.x)
                              && min(a.y,b.y) <= p.y && p.y <= max(a.y,b.y);
}
double dist(const Pt& a, const Pt& b) { return hypot((double)(a.x-b.x), (double)(a.y-b.y)); }

12.1 Overflow, again (the top cause of wrong answers)

// 1. Multiplication promoted AFTER the multiply
int a = 1e5, b = 1e5;
long long bad = a * b;            // int*int computed in int → overflow
long long ok  = 1LL * a * b;      // promote first

// 2. accumulate's type comes from the INITIAL VALUE
long long sum = accumulate(v.begin(), v.end(), 0);    // ✗ sums in int
long long sum2 = accumulate(v.begin(), v.end(), 0LL); // ✓

// 3. (lo + hi) / 2 overflow in binary search → lo + (hi - lo) / 2
// 4. INT_MAX + something in relaxations → use a large but safe INF
const long long INF = 1e18;                 // adding two INF = 2e18 < 9.2e18 (max ll)
// ✗ if (dist[u] + w < dist[v]) with dist[u] == INT_MAX → overflow, WAs
// ✓ either check dist[u] != INF first, or use ll with INF = 4e18 and never overflow by capping

// 5. Counting: n(n-1)/2 for n = 1e5 is 5e9 → int is destroyed, ll is fine.
// 6. Modular multiplication of two values near 1e9: a*b ≈ 1e18 → needs ll (fits), but
//    squaring near 1e18 wraps — use __int128 for modular products when MOD > 2^31.
__int128 prod = (__int128)a * b % MOD;      // GCC/Clang extension, fine on the judge

    for (int l = 0; l + len - 1 < n; ++l) { int r = l + len - 1; }

// Recursion: the three questions to answer before coding
// 1) What is the base case?            (nullptr / l >= r / index == n)
// 2) What does the recursive call return? (an ANSWER, not a side effect if possible)
// 3) Is depth a risk?                  (10^5-deep DFS on a chain → stack overflow)

// Deep recursion fix: explicit stack
void dfsIter(int src) {
    vector<int> st{src};
    while (!st.empty()) {
        int u = st.back(); st.pop_back();
        if (visited[u]) continue;
        visited[u] = true;
        for (int v : adj[u]) if (!visited[v]) st.push_back(v);
    }
}
