3 Strings and characters

C++ strings are mutable, indexable byte arrays with a rich API — much closer to Python lists than Python strings.

3.1 The operations you will type most

string s = "hello world";
int n = (int)s.size();                 // 11  — cast to int, size() is unsigned

// Access and mutation (C++ strings ARE mutable)
char c = s[0];                         // 'h'  — no bounds check
s[0] = 'H';                            // "Hello world"
char last = s.back();                  // 'd'
s.front(); s.at(1);                    // 'e' (at() checks range, throws)

// Slicing: substr(start, LENGTH)
s.substr(0, 5)                         // "Hello"
s.substr(6)                            // "world"   (to the end)
s.substr(6, 3)                         // "wor"

// Searching
s.find("world")                        // 6        ; npos if absent
s.find('o', 5)                         // start searching from index 5
s.rfind('o')                           // 7        ; reverse search
s.find_first_of("aeiou")               // 1
s.find_last_not_of('d')                // 9
if (s.find("xyz") == string::npos) /* not found */;

// Modifying
s += "!";  s.append("?");  s.push_back('.');
s.insert(0, ">> ");                     // insert at position
s.erase(0, 3);                          // erase 3 chars from index 0
s.erase(s.begin());                     // erase first char (iterator form)
s.pop_back();
s.replace(0, 5, "Howdy");               // replace 5 chars at 0
s.clear();

// Comparisons and conversion
string a = "abc", b = "abd";
a == b; a < b;                          // lexicographic, like Python
s.compare("Hello") == 0;                // 0 = equal
int   num = stoi("42");                 // 42      ; stoll, stod, stof
long long big = stoll("9000000000");
string str = to_string(3.14);           // "3.140000"
string sv = to_string(255);             // "255"

// Reverse, sort, unique characters
reverse(s.begin(), s.end());
sort(s.begin(), s.end());               // in-place character sort
s.erase(unique(s.begin(), s.end()), s.end());   // dedupe SORTED string

// Iterate
for (char ch : s) { }
for (int i = 0; i < (int)s.size(); ++i) { }
for (auto it = s.begin(); it != s.end(); ++it) { }

Three classic string traps: (1) s.substr(i, len) — the second parameter is a length; s.substr(i, j-i) for [i, j). (2) s.find(c) returns string::npos (a huge unsigned value), not −1 — always compare against npos; and never write if (s.find(c) >= 0) (always true, unsigned). (3) tolower()/toupper() take and return int: assign back as char, and pass unsigned char for safety with negative char values.

4 vector, arrays and grids

The workhorse container: growth rules, reserve, 2D grids, and the iterator model you must understand.

4.1 Creating and using a vector

#include <bits/stdc++.h>
using namespace std;

vector<int> v;                        // empty
vector<int> a(5);                     // 5 zeros:      {0,0,0,0,0}
vector<int> b(5, -1);                 // 5 copies of -1
vector<int> c{1, 2, 3};               // initialiser list
vector<int> d(c);                     // copy (O(n))
vector<int> e(c.begin(), c.end());    // from iterators
int arr[] = {1,2,3};
vector<int> f(arr, arr + 3);          // from a C array

// Sizing and capacity
v.size(); v.empty(); v.capacity();
v.resize(10);                         // grow (zero-fills) or shrink
v.resize(10, 7);                      // fill new slots with 7
v.reserve(1000);                      // preallocate: avoids repeated reallocation
v.shrink_to_fit();                    // release spare capacity (rarely needed)

// Access
v[0]; v.at(0); v.front(); v.back();
int* p = v.data();                    // raw pointer to the buffer

// Add / remove
v.push_back(9);                       // amortised O(1)
v.pop_back();                         // O(1)
v.insert(v.begin() + 1, 42);          // O(n) — shifts the tail
v.insert(v.end(), 3, 5);              // insert 3 copies of 5 at the end
v.erase(v.begin());                   // O(n)
v.erase(v.begin() + 1, v.begin() + 3); // erase a range [1,3)
v.assign(4, 0);                       // replace contents with four 0s
v.clear();                            // size 0, capacity kept

// Iterators (the general way to talk about positions)
v.begin(); v.end();                   // [begin, end) — half-open
v.rbegin(); v.rend();                 // reverse
for (auto it = v.begin(); it != v.end(); ++it) cout << *it;
int pos = lower_bound(v.begin(), v.end(), 5) - v.begin();   // index from iterator

4.2 2D vectors and grids — the LeetCode staple

// ---- construction -------------------------------------------------------
int m = 3, n = 4;
vector<vector<int>> g(m, vector<int>(n, 0));      // 3×4 of zeros  ★ the standard form
vector<vector<int>> g2(m);                          // 3 empty rows
for (auto& row : g2) row.assign(n, 0);              // then fill

// from an initialiser list
vector<vector<int>> grid = {{1,2,3},{4,5,6},{7,8,9}};

// ---- traversal ----------------------------------------------------------
int R = grid.size(), C = grid[0].size();
for (int i = 0; i < R; ++i)
    for (int j = 0; j < C; ++j)
        /* grid[i][j] */;

// ---- neighbour moves: two idioms ---------------------------------------
const int di[4] = {-1, 1, 0, 0};       // up, down, left, right
const int dj[4] = {0, 0, -1, 1};
for (int d = 0; d < 4; ++d) {
    int ni = i + di[d], nj = j + dj[d];
    if (ni < 0 || ni >= R || nj < 0 || nj >= C) continue;   // bounds check FIRST
    /* visit grid[ni][nj] */
}
// 8-directional (king moves)
const int dx8[8] = {-1,-1,-1, 0, 0, 1, 1, 1};
const int dy8[8] = {-1, 0, 1,-1, 1,-1, 0, 1};

// ---- in-place 90° rotation of an n×n matrix ----------------------------
for (int i = 0; i < n; ++i)
    for (int j = i + 1; j < n; ++j)
        swap(g[i][j], g[j][i]);          // transpose
for (auto& row : g) reverse(row.begin(), row.end());   // then mirror → rotated clockwise

// ---- spiral traversal skeleton -----------------------------------------
int top = 0, bottom = R - 1, left = 0, right = C - 1;
while (top <= bottom && left <= right) {
    for (int j = left; j <= right; ++j) visit(g[top][j]);      ++top;
    for (int i = top; i <= bottom; ++i) visit(g[i][right]);    --right;
    if (top <= bottom) { for (int j = right; j >= left; --j) visit(g[bottom][j]); --bottom; }
    if (left <= right) { for (int i = bottom; i >= top; --i) visit(g[i][left]);  ++left;  }
}

// ---- flattening a grid into 1D (useful for DSU/visited arrays) ---------
int id = i * C + j;          int i2 = id / C, j2 = id % C;
Grid checklist: always compute R/C once; guard grid.empty() before grid[0]; do bounds checks before indexing; prefer vector<vector<int>> for clarity and a flat vector<int> of size R·C when you need speed (better cache locality). For visited, a 2D vector<char> is often faster to initialise than vector<bool>.

5.3 unordered_map / unordered_set in practice

unordered_map<string, int> cnt;              // hash table, O(1) average
unordered_map<int, vector<int>> groups;      // adjacency / grouping
unordered_set<int> seen;

// Insert / access
cnt["apple"]++;                              // default-constructs 0 then increments ★
cnt.insert({"banana", 3});                   // does NOT overwrite an existing key
cnt.emplace("cherry", 5);                    // constructs in place (faster)
cnt["date"] = 7;                             // insert-or-assign

// Lookup — three ways, know the difference
if (cnt.count("apple")) { }                  // 1 = present, 0 = absent
auto it = cnt.find("apple");
if (it != cnt.end()) cout << it->second;     // ★ no double lookup
int v = cnt.at("apple");                     // throws std::out_of_range if absent
int safe = cnt["missing"];                   // ⚠ INSERTS a 0 and returns it

// Iterate (order is arbitrary — do not rely on it)
for (const auto& [key, value] : cnt) { }
for (auto it = cnt.begin(); it != cnt.end(); ++it) { it->first; it->second; }

// Erase — the safe loop
for (auto it = cnt.begin(); it != cnt.end(); ) {
    if (it->second == 0) it = cnt.erase(it);   // erase returns the next iterator
    else ++it;
}
cnt.erase("apple");                            // by key, returns count erased

// Size, reserve, clear
cnt.size(); cnt.empty(); cnt.clear();
cnt.reserve(10000);                            // pre-size buckets: avoids rehashing
cnt.max_load_factor(0.7);                      // tune if you see rehash spikes

// Sets (same API minus values)
seen.insert(5); seen.count(5); seen.erase(5);
seen.find(5) != seen.end();
The operator[] trap: m[key] inserts a default value if the key is missing. Accidentally using it in a read path grows your map (and is a common source of wrong answers in loops). Use find()/count()/at() to read, and reserve [] for writing.

5.4 Ordered map / set / multiset — the secret weapon

map<string, int> m;        // sorted by key, O(log n)
set<int> s;                // sorted unique keys
multiset<int> ms;          // sorted, duplicates allowed

// Ordered containers keep the keys sorted → these idioms win problems
*s.begin();                 // minimum key            (O(1) via cached begin)
*prev(s.end());             // maximum key             (O(log n) or O(1) for map)
auto it = s.lower_bound(5);  // first element >= 5
auto it2 = s.upper_bound(5); // first element >  5
s.count(5);  ms.count(5);    // set: 0/1 · multiset: number of copies

// Predecessor / successor — the reason to choose ordered containers
auto it3 = s.lower_bound(x);
if (it3 != s.begin()) { auto pred = prev(it3); /* largest < x */ }
if (it3 != s.end())   { /* smallest >= x */ }

// Iterate in sorted order
for (int v : s) cout << v << ' ';
for (const auto& [k, v] : m) cout << k << ":" << v << ' ';

// Multiset erase carefully
ms.erase(3);                       // ⚠ erases ALL copies
ms.erase(ms.find(3));              // erases ONE copy ← the usual intent

// Sliding-window median with two multisets (classic hard problem)
multiset<int> low, high;           // low: lower half (max at rbegin), high: upper half
// maintain low.size() == high.size() or low.size() == high.size()+1
// median = *low.rbegin()  (odd)  or (*low.rbegin() + *high.begin()) / 2.0  (even)

// Ordered map as an LRU/cheapest-first structure
map<int, int> idxToVal;            // also: count of distinct keys in a range, etc.
Price/performance rule: unordered_set is ~3–5× faster than set in wall-clock time, but set unlocks lower_bound, sorted iteration, min/max and predecessor/successor in O(log n). When the problem needs “next greater / nearest value / ordered sweep”, the ordered container is not a luxury — it is the algorithm.

5.5 stack, queue, deque

// ---- stack (LIFO) -------------------------------------------------------
stack<int> st;                  // adapters over deque by default
st.push(1); st.top(); st.pop(); st.size(); st.empty();
// Faster alternative: vector as a stack (better cache locality)
vector<int> st2; st2.push_back(1); st2.back(); st2.pop_back();

// ---- queue (FIFO) — the BFS container -----------------------------------
queue<int> q;
q.push(1); q.front(); q.back(); q.pop(); q.size(); q.empty();

// ---- deque: O(1) at both ends, random access ----------------------------
deque<int> dq{1,2,3};
dq.push_front(0); dq.push_back(4);
dq.pop_front(); dq.pop_back();
dq[0]; dq.front(); dq.back();

// ★ Monotonic deque — sliding window maximum in O(n) (ch. 11 pattern)
vector<int> maxSlidingWindow(const vector<int>& a, int k) {
    deque<int> idx;                                  // indices, values DECREASING
    vector<int> res;
    for (int i = 0; i < (int)a.size(); ++i) {
        while (!idx.empty() && idx.front() <= i - k) idx.pop_front();   // drop expired
        while (!idx.empty() && a[idx.back()] <= a[i]) idx.pop_back();   // keep monotonic
        idx.push_back(i);
        if (i >= k - 1) res.push_back(a[idx.front()]);                  // window max
    }
    return res;
}

// Monotonic stack — next greater element in O(n)
vector<int> nextGreater(const vector<int>& a) {
    int n = a.size();
    vector<int> res(n, -1);
    stack<int> st;                                    // indices, decreasing values
    for (int i = 0; i < n; ++i) {
        while (!st.empty() && a[st.top()] < a[i]) { res[st.top()] = a[i]; st.pop(); }
        st.push(i);
    }
    return res;
}

5.6 priority_queue — the heap

// ⚠ Default is a MAX-heap (Python's heapq is a min-heap — the opposite!)
priority_queue<int> pq;                       // top() = largest
priority_queue<int, vector<int>, greater<int>> minpq;   // top() = smallest

pq.push(3); pq.push(1); pq.push(4);
pq.top();      // 4
pq.pop();      // removes the top
pq.size(); pq.empty();

// Custom comparator: "greater" means the SMALLER element has higher priority
auto cmp = [](const pair<int,int>& a, const pair<int,int>& b) {
    return a.second > b.second;               // min-heap by .second
};
priority_queue<pair<int,int>, vector<pair<int,int>>, decltype(cmp)> pq2(cmp);

// Pairs: comparison is lexicographic (first, then second) — no comparator needed
priority_queue<pair<int,int>> p;              // max by first, then second

// Dijkstra with a priority queue (lazy-deletion style)
vector<long long> dist(n, LLONG_MAX);
priority_queue<pair<long long,int>, vector<pair<long long,int>>, greater<>> heap;
dist[src] = 0; heap.push({0, src});
while (!heap.empty()) {
    auto [d, u] = heap.top(); heap.pop();
    if (d > dist[u]) continue;                // stale entry — skip
    for (auto [v, w] : adj[u])
        if (d + w < dist[v]) { dist[v] = d + w; heap.push({dist[v], v}); }
}

// Top-K pattern: keep a MIN-heap of size k → O(n log k) time, O(k) memory
priority_queue<int, vector<int>, greater<int>> top;   // smallest of the current k on top
for (int x : nums) {
    top.push(x);
    if ((int)top.size() > k) top.pop();
}   // top now holds the k largest values
The comparator trap that costs the most submissions. In priority_queue, the comparator answers “should a come after b?” — the element the comparator calls “less” ends up on top. So greater<int> gives a min-heap and less<int> a max-heap. Also, a custom comparator must be a strict weak ordering (irreflexive, transitive): writing >= instead of > can make the heap corrupt the array with no error message.
