16.1 Everything is a heap object (and that costs memory)

import sys, tracemalloc
sys.getsizeof(0), sys.getsizeof(10**100)      # 28 … grows with magnitude
sys.getsizeof("") , sys.getsizeof("abc")      # 49, 52 bytes

tracemalloc.start()

# __slots__: drop the per-instance __dict__ → big savings for many small objects
class PointSlots:
    __slots__ = ("x", "y")
    def __init__(self, x, y): self.x, self.y = x, y
class PointDict:
    def __init__(self, x, y): self.x, self.y = x, y

sys.getsizeof(PointSlots(1, 2)) + sys.getsizeof(PointSlots.__dict__)
sys.getsizeof(PointDict(1, 2)) + sys.getsizeof(PointDict(1, 2).__dict__)   # far bigger

# Weak references: keep a cache without preventing collection
import weakref
class Node: ...
n = Node()
r = weakref.ref(n)
r() is n                 # alive
del n
r()                      # None — no leak
cache = weakref.WeakValueDictionary()   # entries vanish with their values
# Use for: caches, observer registries, parent back-pointers, avoiding cycles

16.2 Reference counting + the cycle collector

import sys, gc

a = []
b = [a]
a.append(b)                 # cycle!
sys.getrefcount(a)          # includes the temporary arg reference — subtract 1 mentally
del a, b
gc.collect()                # returns the number of unreachable objects collected
gc.get_count()              # (gen0, gen1, gen2) counters
gc.get_threshold()          # default (700, 10, 10) — tune for latency-sensitive apps
gc.set_threshold(50_000, 50, 50)   # fewer, larger collections ── measure first!
gc.freeze()                 # move long-lived objects out of GC's view (after startup)
gc.disable()                # some servers do this and manage memory themselves — risky

# __del__ is NOT a destructor you should rely on:
class Noisy:
    def __del__(self):
        print("collected")      # timing is unspecified; may never run at interpreter exit
# Prefer explicit cleanup: context managers, close(), or weakref.finalize
import weakref
finalizer = weakref.finalize(obj, cleanup_fn, arg)   # runs at collection, deterministic-ish
finalizer.detach()                                    # cancel it

17.2 The optimisation checklist (in order of payoff)

  1. Fix the algorithm. O(n²) → O(n log n) or O(n) beats every micro-trick. Check chapter 5’s complexity table: in list → in set; list.pop(0) → deque.popleft(); nested loop over two collections → one dict lookup.
  2. Do less work. Cache (@cache, HTTP caching, materialised views), short-circuit (any/genexp), avoid re-computing what you already have.
  3. Move work into C. sum/min/max/sorted/join/dict comprehension over Python-level loops; re over manual string scanning; then NumPy/polars/Arrow/orjson for bulk data.
  4. Change the data structure. list → set/dict, list of dicts → list of tuples or NumPy array, Python dict → frozenset, array, or a column store.
  5. Stream instead of materialise. Generators, chunked reads, islice, database cursors — lowers memory and improves cache locality.
  6. Batch I/O. One query with WHERE id IN (…) instead of 100 queries; executemany; bulk API endpoints; asyncio.gather.
  7. Parallelise only after 1–6 (chapter 15).
  8. Compile the hot 5%: Cython, mypyc, Numba (@njit), PyO3/Rust extension, or run under a faster interpreter (PyPy for pure-Python loops — no C-extension support).
# The classics, measured
# 1. membership    O(n) → O(1)
if item in set(big_list): ...                    # build the set ONCE, outside the loop

# 2. string building
parts = []; ... ; "".join(parts)                 # not s += x in a loop (O(n²))

# 3. local variable lookup beats global/attribute lookup
def total(xs):
    s = 0
    for x in xs: s += x                          # locals are the fastest namespace
    return s
sum(xs)                                          # …but a builtin in C is faster still

# 4. avoid repeated attribute lookups in hot loops
append = result.append                          # bind the method once
for x in data: append(transform(x))

# 5. comprehensions are faster than explicit loops + append
[x * 2 for x in xs]

# 6. NumPy: vectorise — the single biggest win for numeric code
# slow:  total = sum(x * 2 for x in arr)
# fast:  total = (arr * 2).sum()
import numpy as np
a = np.arange(1_000_000)
%timeit a.sum()          # in IPython/Jupyter: %timeit, %prun, %%memit (memory_profiler)

# 7. avoid pandas row-wise apply — vectorise or use polars
# slow:  df.apply(lambda r: r.a + r.b, axis=1)
# fast:  df["a"] + df["b"]      /  pl.col("a") + pl.col("b")

# 8. serialisation: json → orjson/msgspec for big payloads (2–10×)
import orjson
orjson.dumps(data)          # bytes; loads() is far faster than json.loads

# 9. startup time: lazy imports, fewer imports at module level
$ python -X importtime app.py 2> import_log.txt

# 10. the interpreter itself improved: 3.11/3.12 are 10–60% faster than 3.10 on
#     typical code — sometimes "upgrade Python" is the whole optimisation.
Know what is faster, roughly (CPython 3.12+, relative, illustrative): builtin C function 1× · comprehension 2–3× slower than the equivalent builtin · plain Python loop 5–20× slower than the builtin · NumPy on 1 M elements 50–500× faster than a Python loop · a Rust/C extension another 2–10× on top. The pattern: push loops down into C.

18.1 The test pyramid, and what to actually test

# tests/test_orders.py — pytest style
import pytest
from my_pkg.orders import apply_discount, OrderError

# --- plain asserts, no boilerplate ---------------------------------------
def test_apply_discount_basic():
    assert apply_discount(100, "SAVE10") == 90

# --- parametrize: one test, many cases ----------------------------------
@pytest.mark.parametrize(
    ("amount", "code", "expected"),
    [
        (100, "SAVE10", 90),
        (0,   "SAVE10", 0),
        (100, None,     100),
        (100, "FLAT50", 50),
    ],
)
def test_discounts(amount, code, expected):
    assert apply_discount(amount, code) == expected

# --- exceptions ----------------------------------------------------------
def test_discount_rejects_negative():
    with pytest.raises(OrderError, match="negative"):
        apply_discount(-5, "SAVE10")

# --- approx for floats ---------------------------------------------------
def test_float_math():
    assert 0.1 + 0.2 == pytest.approx(0.3)

18.4 Docstrings — the format that tooling understands

def transfer(src: "Account", dst: "Account", amount: float) -> None:
    """Move money between two accounts atomically.

    Args:
        src: Account to debit. Must have sufficient balance.
        dst: Account to credit.
        amount: Positive value in the account currency.

    Raises:
        ValueError: If amount is not positive.
        InsufficientFunds: If src.balance < amount.
        DatabaseError: If the transaction cannot be committed.

    Examples:
        >>> transfer(alice, bob, 100.0)
        >>> bob.balance
        100.0
    """
