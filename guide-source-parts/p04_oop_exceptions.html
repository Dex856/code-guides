8 Object-oriented Python and the data model

Classes, dunder methods, inheritance & MRO, dataclasses, protocols and enums — done the Pythonic way.

8.1 Classes, instances, and where attributes live

class Account:
    kind = "savings"                      # CLASS attribute — shared by all instances
    __slots__ = ("owner", "balance")      # optional: no __dict__, ~40% less memory

    def __init__(self, owner: str, balance: float = 0.0):
        self.owner = owner                # INSTANCE attributes (per object)
        self.balance = balance

    def deposit(self, amount: float) -> None:
        if amount <= 0:
            raise ValueError("amount must be positive")
        self.balance += amount

    def __repr__(self) -> str:            # unambiguous, for developers
        return f"Account({self.owner!r}, {self.balance})"

    def __str__(self) -> str:             # readable, for users (falls back to __repr__)
        return f"{self.owner}: ₹{self.balance:,.2f}"

a = Account("Ada", 100)
a.deposit(50)
print(a)                                  # Ada: ₹150.00
a.kind, Account.kind                      # 'savings' both ways
a.kind = "current"                        # creates an INSTANCE attribute that shadows the class one
del a.kind                                # reveals the class attribute again

# Attribute lookup order: instance __dict__ → type's MRO dicts → __getattr__ → AttributeError
vars(a)          # instance dict (not available with __slots__)
dir(a)           # everything reachable, for the REPL
hasattr(a, "x"), getattr(a, "x", None), setattr(a, "x", 1)

from dataclasses import dataclass
from functools import total_ordering

@total_ordering
class Money:
    __slots__ = ("amount", "currency")

    def __init__(self, amount: int, currency: str = "INR"):   # store minor units (paise)
        self.amount, self.currency = amount, currency

    def __repr__(self):  return f"Money({self.amount}, {self.currency!r})"
    def __str__(self):   return f"{self.currency} {self.amount / 100:,.2f}"
    def __eq__(self, other):    return (self.amount, self.currency) == (other.amount, other.currency) if isinstance(other, Money) else NotImplemented
    def __hash__(self): return hash((self.amount, self.currency))
    def __lt__(self, other): return self.amount < other.amount
    def __add__(self, other):   return Money(self.amount + other.amount, self.currency)
    def __mul__(self, n):       return Money(self.amount * n, self.currency)
    def __bool__(self): return self.amount != 0

Money(1000) + Money(250)      # Money(1250, 'INR')  →  INR 12.50
bool(Money(0))                # False
sorted([Money(500), Money(100)])[0]
Return NotImplemented (the sentinel, not NotImplementedError) from a binary dunder when you cannot handle the other type — that lets Python try the reflected operation (other.__radd__) instead of raising. It is how 2 "a" … sorry, how "a" * 2 and 2 * "a" both work.

8.3 Inheritance, MRO and super()

class Base:
    def __init__(self, **kwargs):
        super().__init__(**kwargs)          # cooperative multiple inheritance
        self.ready = True

class Loggable:
    def log(self, msg): print(f"[log] {msg}")

class Cacheable:
    _cache: dict = {}
    def cached(self, key, factory):
        if key not in self._cache:
            self._cache[key] = factory()
        return self._cache[key]

class Service(Loggable, Cacheable, Base):   # mixins first, concrete base last
    def run(self):
        self.log("starting")
        return self.cached("k", lambda: 42)

Service.__mro__          # (Service, Loggable, Cacheable, Base, object)
s = Service(); s.run()

# Abstract base classes: enforce an interface at instantiation time
from abc import ABC, abstractmethod
class Repository(ABC):
    @abstractmethod
    def get(self, key: str) -> dict: ...
    @property
    @abstractmethod
    def name(self) -> str: ...

# Repository()  → TypeError: Can't instantiate abstract class

# Duck typing without inheritance: Protocol (structural typing, ch. 13)
from typing import Protocol, runtime_checkable
@runtime_checkable
class Closeable(Protocol):
    def close(self) -> None: ...
isinstance(socket_like, Closeable)          # True if it has .close()

8.4 Dataclasses (and when to skip classes entirely)

from dataclasses import dataclass, field, asdict, replace

@dataclass(slots=True)                       # slots=True → less memory, faster attr access
class Product:
    sku: str
    price: float
    tags: list[str] = field(default_factory=list)    # ← never use a bare [] here
    active: bool = True

    def __post_init__(self):                 # validation after generated __init__
        if self.price < 0:
            raise ValueError("price must be >= 0")

p = Product("A1", 99.5, ["sale"])
p == Product("A1", 99.5, ["sale"])           # True — __eq__ generated
replace(p, price=89.0)                       # copy with a change
asdict(p)                                    # → plain dict (for JSON)

@dataclass(frozen=True)                      # immutable + hashable → usable in sets/dict keys
class Point:
    x: int
    y: int

# Ordering and safety switches
@dataclass(order=True)                       # generates __lt__ etc. via field order
class Version:
    major: int
    minor: int

# Alternatives: NamedTuple (tuple semantics), attrs (mature extras),
# pydantic (runtime validation + parsing from JSON), msgspec (fastest).
from pydantic import BaseModel, Field
class User(BaseModel):
    id: int
    email: str = Field(pattern=r".+@.+")
    nickname: str | None = None
User.model_validate_json('{"id": 1, "email": "a@b.c"}')   # validated, typed
User(id="2", email="x@y.z")                                # coerces "2" → 2 or raises

Enums — replace magic strings

from enum import Enum, StrEnum, IntEnum, auto, Flag

class Status(StrEnum):            # str-compatible: works in f-strings, JSON, SQL
    PENDING = auto()
    ACTIVE = "active"             # explicit value
    BANNED = "banned"

Status.ACTIVE == "active"          # True — StrEnum subclass of str
status = Status("active")          # lookup by value → Status.ACTIVE
Status.ACTIVE.name, Status.ACTIVE.value

class Perm(Flag):                  # bitwise combinations
    READ, WRITE, EXEC = auto(), auto(), auto()
perms = Perm.READ | Perm.WRITE
Perm.READ in perms                 # True

@dataclass
class Order: status: Status = Status.PENDING
Order().status is Status.PENDING

Properties, static and class methods

class Temperature:
    def __init__(self, celsius: float): self._c = celsius

    @property                                  # computed, read-only attribute
    def celsius(self) -> float: return self._c

    @celsius.setter                            # validate on assignment
    def celsius(self, v: float):
        if v < -273.15: raise ValueError("below absolute zero")
        self._c = v

    @property
    def fahrenheit(self) -> float: return self._c * 9 / 5 + 32

    @staticmethod                              # no access to self/cls — a namespaced function
    def from_kelvin(k: float) -> "Temperature": return Temperature(k - 273.15)

    @classmethod                               # alternative constructor (inherits properly)
    def from_fahrenheit(cls, f: float) -> "Temperature": return cls((f - 32) * 5 / 9)

t = Temperature(25); t.fahrenheit            # 77.0
Temperature.from_fahrenheit(77.0).celsius    # 25.0
t.celsius = -300                             # ValueError

9.2 try / except / else / finally — the full flow

import json, logging

log = logging.getLogger(__name__)

def load_config(path: str) -> dict:
    try:
        with open(path, encoding="utf-8") as fh:      # risky: file may not exist
            raw = fh.read()
    except FileNotFoundError:
        log.warning("no config at %s, using defaults", path)
        return {}
    except PermissionError as e:
        raise ConfigError(f"cannot read {path}") from e        # add context, keep cause
    else:
        try:
            return json.loads(raw)                                 # parse errors are different
        except json.JSONDecodeError as e:
            raise ConfigError(f"{path} is not valid JSON: {e}") from e
    finally:
        log.debug("load_config finished")                          # always runs

# Round-trip: catch → inspect → decide
try:
    value = int(user_input)
except ValueError:
    value = 0
except (KeyError, IndexError) as e:        # several types
    log.error("bad structure: %r", e)
except Exception:                           # last resort: log & re-raise, never silently pass
    log.exception("unexpected failure")
    raise
else:
    use(value)                              # only if no exception
finally:
    cleanup()

# EAFP vs LBYL
# EAFP (Pythonic):         # LBYL (risky under concurrency)
try:                       # if key in d:
    v = d[key]             #     v = d[key]
except KeyError:           # else:
    v = default            #     v = default

# Re-raising with context
try:
    api_call()
except requests.Timeout as e:
    raise ServiceUnavailable("upstream timed out after 5s") from e
# On the traceback you'll see BOTH: "The above exception was the direct cause..."
# Use `raise ... from None` to suppress the original when it is just noise.
Three anti-patterns that hide real bugs: (1) except Exception: pass — swallows everything, including typos; (2) try: big_block_of_code() — you cannot tell which line failed; (3) catching an exception just to raise SomeError(str(e)) — this loses the original traceback; always use raise NewError(...) from e.

9.4 Context managers — guaranteed cleanup

# --- Using what exists ----------------------------------------------------
from contextlib import (contextmanager, closing, suppress, redirect_stdout,
                        ExitStack, nullcontext, chdir, AsyncExitStack)
import tempfile, sqlite3, io, os, time

with open("data.txt", encoding="utf-8") as fh:            # closes even on exception
    data = fh.read()

with suppress(FileNotFoundError):                          # ignore a specific error, cleanly
    os.remove("maybe-there.tmp")

with tempfile.TemporaryDirectory() as tmp:                 # auto-cleanup
    path = os.path.join(tmp, "scratch.bin")

with sqlite3.connect("app.db") as conn:                    # commits on success, rolls back on error
    conn.execute("insert into t values (?)", (1,))

buf = io.StringIO()
with redirect_stdout(buf):
    print("hello")                                         # captured into buf

# --- Writing your own (generator version) --------------------------------
@contextmanager
def timed_block(label: str):
    t0 = time.perf_counter()
    try:
        yield                                              # with-block runs here
    finally:
        print(f"{label}: {time.perf_counter()-t0:.3f}s")   # always

# --- Writing your own (class version — reusable, inspectable) -------------
class Transaction:
    def __init__(self, conn):
        self.conn = conn
    def __enter__(self):
        self.conn.execute("BEGIN")
        return self.conn
    def __exit__(self, exc_type, exc, tb):
        if exc_type is None:
            self.conn.execute("COMMIT")
        else:
            self.conn.execute("ROLLBACK")                   # return None → exception propagates
        return False

# --- Managing many at once (nested with-blocks, flattened) ----------------
with ExitStack() as stack:
    files = [stack.enter_context(open(p)) for p in paths]
    conn = stack.enter_context(sqlite3.connect("app.db"))
    # everything closed/rolled back in reverse order at the end of the block
Design tip: state-changing helpers are the best candidates for context managers — chdir(), pushd, a DB transaction, a temporary feature flag, a mock, a stdout redirect. If you find yourself writing “remember to close/reset this later” in a comment, write a context manager instead.
