19 Building real projects

Architecture that survives growth, plus three starter projects you can ship this weekend.

19.1 Architecture: keep the domain independent

Project sizeStructure that fits
Script (< 200 lines)One file + if __name__ == "__main__". Do not over-engineer.
Small tool / packagesrc/pkg/ with a few modules, tests/, pyproject, py.typed.
CLI applicationtyper/argparse entry point → services module → adapters. Config via env + pydantic-settings.
Web APIrouters/ → services/ → repositories/ + models/ (pydantic) + db/ (SQLAlchemy) + core/ (config, security).
Data pipelineingest/ → transform/ (pure functions) → load/ + orchestration (Prefect/Dagster/Airflow) + data contracts.
LibraryNarrow public API in __init__.py, typed, documented, semantic versioning, CI matrix.

19.2 Project 1 — a CLI tool that people will actually install

# src/tasky/__init__.py        src/tasky/cli.py        src/tasky/store.py
# pyproject.toml: [project.scripts] tasky = "tasky.cli:app"

# --- store.py: the domain/data layer (pure, testable) --------------------
import json, sqlite3
from dataclasses import dataclass, asdict
from pathlib import Path

DB = Path.home() / ".tasky.db"

@dataclass(slots=True)
class Task:
    id: int | None
    title: str
    done: bool = False
    tags: list[str] = None

class Store:
    def __init__(self, path: Path = DB):
        self.con = sqlite3.connect(path)
        self.con.row_factory = sqlite3.Row
        self.con.execute("""CREATE TABLE IF NOT EXISTS tasks(
            id INTEGER PRIMARY KEY, title TEXT NOT NULL,
            done INTEGER DEFAULT 0, tags TEXT DEFAULT '')""")

    def add(self, title: str, tags: list[str] | None = None) -> int:
        cur = self.con.execute("INSERT INTO tasks(title, tags) VALUES (?, ?)",
                               (title, ",".join(tags or [])))
        self.con.commit()
        return cur.lastrowid

    def list(self, only_open: bool = False) -> list[Task]:
        sql = "SELECT * FROM tasks" + (" WHERE done = 0" if only_open else "") + " ORDER BY id"
        return [Task(r["id"], r["title"], bool(r["done"]), r["tags"].split(",") if r["tags"] else [])
                for r in self.con.execute(sql)]

    def complete(self, task_id: int) -> bool:
        cur = self.con.execute("UPDATE tasks SET done = 1 WHERE id = ?", (task_id,))
        self.con.commit()
        return cur.rowcount > 0

# --- cli.py: the adapter layer (thin) -----------------------------------
import typer
from rich.console import Console
from rich.table import Table
from .store import Store

app = typer.Typer(help="Tiny task manager.")
console = Console()

@app.command()
def add(title: str, tag: list[str] = typer.Option(None, "--tag", "-t")):
    """Add a task."""
    new_id = Store().add(title, tag)
    console.print(f"[green]added[/] #{new_id}: {title}")

@app.command("list")
def list_tasks(open_only: bool = typer.Option(False, "--open", "-o")):
    """List tasks."""
    tasks = Store().list(open_only)
    table = Table("id", "title", "done", "tags")
    for t in tasks:
        table.add_row(str(t.id), t.title, "✓" if t.done else "", ",".join(t.tags or []))
    console.print(table)

@app.command()
def done(task_id: int):
    """Mark a task complete."""
    if not Store().complete(task_id):
        raise typer.Exit(code=1)          # nonzero exit code for scripts/CI
    console.print(f"[green]completed[/] #{task_id}")

if __name__ == "__main__":
    app()

# Ship it:
# $ uv build && uv publish            (or: uv tool install --from . tasky)
# $ uvx --from tasky tasky list

19.3 Project 2 — a web API with FastAPI

# app/main.py
from contextlib import asynccontextmanager
from fastapi import FastAPI, HTTPException, Depends, status
from pydantic import BaseModel, Field
import sqlite3, os

DB_PATH = os.environ.get("APP_DB", "app.db")

def get_db():
    con = sqlite3.connect(DB_PATH, check_same_thread=False)
    con.row_factory = sqlite3.Row
    try:
        yield con
    finally:
        con.close()

class ItemIn(BaseModel):
    name: str = Field(min_length=1, max_length=80)
    price: float = Field(gt=0)

class Item(ItemIn):
    id: int

@asynccontextmanager
async def lifespan(app: FastAPI):        # startup/shutdown (replaces @app.on_event)
    con = sqlite3.connect(DB_PATH)
    con.execute("CREATE TABLE IF NOT EXISTS items(id INTEGER PRIMARY KEY, name TEXT, price REAL)")
    con.commit(); con.close()
    yield

app = FastAPI(title="Catalog API", version="1.0.0", lifespan=lifespan)

@app.get("/health")
def health() -> dict[str, str]:
    return {"status": "ok"}

@app.get("/items", response_model=list[Item])
def list_items(limit: int = 20, offset: int = 0, db=Depends(get_db)):
    rows = db.execute("SELECT * FROM items LIMIT ? OFFSET ?", (limit, offset)).fetchall()
    return [dict(r) for r in rows]

@app.post("/items", response_model=Item, status_code=status.HTTP_201_CREATED)
def create_item(item: ItemIn, db=Depends(get_db)):
    cur = db.execute("INSERT INTO items(name, price) VALUES (?, ?)", (item.name, item.price))
    db.commit()
    return Item(id=cur.lastrowid, **item.model_dump())

@app.get("/items/{item_id}", response_model=Item)
def get_item(item_id: int, db=Depends(get_db)):
    row = db.execute("SELECT * FROM items WHERE id = ?", (item_id,)).fetchone()
    if row is None:
        raise HTTPException(status_code=404, detail="item not found")
    return dict(row)

# run:  uv run fastapi dev app/main.py     → interactive docs at /docs (OpenAPI for free)
# tests: from fastapi.testclient import TestClient ; client = TestClient(app)

19.4 Project 3 — a data pipeline that does not break

# pipeline.py — ingest → validate → transform → load, with retries and observability
from dataclasses import dataclass
from pathlib import Path
import csv, json, logging, sqlite3, time
import polars as pl

log = logging.getLogger("pipeline")

@dataclass(slots=True)
class Config:
    source: Path
    db: Path
    batch: int = 5_000

def extract(cfg: Config) -> pl.DataFrame:
    log.info("reading %s", cfg.source)
    return pl.read_csv(cfg.source, try_parse_dates=True)      # lazy: pl.scan_csv for huge files

def validate(df: pl.DataFrame) -> pl.DataFrame:
    bad = df.filter((pl.col("qty") <= 0) | pl.col("sku").is_null())
    if bad.height:
        log.warning("dropping %d invalid rows", bad.height)
    return df.filter((pl.col("qty") > 0) & pl.col("sku").is_not_null())

def transform(df: pl.DataFrame) -> pl.DataFrame:
    return (df.with_columns(
                (pl.col("qty") * pl.col("unit_price")).alias("total"),
                pl.col("sku").str.to_uppercase(),
            )
              .group_by("sku")
              .agg(pl.col("total").sum().round(2).alias("revenue"), pl.len().alias("orders")))

def load(df: pl.DataFrame, cfg: Config) -> int:
    with sqlite3.connect(cfg.db) as con:
        con.execute("CREATE TABLE IF NOT EXISTS revenue(sku TEXT PRIMARY KEY, revenue REAL, orders INT)")
        con.executemany(
            "INSERT INTO revenue(sku, revenue, orders) VALUES (?, ?, ?) "
            "ON CONFLICT(sku) DO UPDATE SET revenue=excluded.revenue, orders=excluded.orders",
            df.iter_rows())
    log.info("loaded %d rows", df.height)
    return df.height

def run(cfg: Config) -> int:
    for attempt in range(3):
        try:
            df = extract(cfg)
            result = transform(validate(df))
            return load(result, cfg)
        except Exception:
            log.exception("attempt %d failed", attempt + 1)
            if attempt == 2:
                raise
            time.sleep(2 ** attempt)
    return 0

if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")
    raise SystemExit(0 if run(Config(Path("sales.csv"), Path("warehouse.db"))) else 1)

# Scheduling in production: cron → `uv run pipeline.py` ; orchestration → Prefect/Dagster/Airflow
# Quality gates: pandera/great-expectations data contracts; write a row count to a metrics table
Every real project needs these five things before it counts as “done”: (1) a pyproject.toml + lock file, (2) tests that run in under 10 seconds, (3) structured logging, (4) a config story (env vars + defaults, no secrets in code), (5) CI that runs format + lint + types + tests on every push. Everything else is optional.
