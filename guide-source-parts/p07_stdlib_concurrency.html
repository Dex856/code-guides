15.2 Threads and thread pools (I/O-bound)

from concurrent.futures import ThreadPoolExecutor, as_completed
import threading, time, queue

# The 90% case: a pool, a list of work, results as they finish
def fetch(url: str) -> str:
    time.sleep(0.2)                      # pretend network call
    return f"body of {url}"

urls = [f"https://ex.com/{i}" for i in range(20)]

with ThreadPoolExecutor(max_workers=8) as pool:          # ~= 2–10× CPU count for I/O
    futures = {pool.submit(fetch, u): u for u in urls}
    for fut in as_completed(futures):
        url = futures[fut]
        try:
            print(url, len(fut.result()))
        except Exception as e:
            print(url, "failed:", e)

results = list(ThreadPoolExecutor(8).map(fetch, urls))    # preserves input order
pool.submit(fn, *args, timeout=5)

# Manual threads when you need long-lived workers
def worker(q: queue.Queue):
    while (item := q.get()) is not None:
        try:
            process(item)
        finally:
            q.task_done()                # unblocks q.join()

q = queue.Queue(maxsize=100)
threads = [threading.Thread(target=worker, args=(q,), daemon=True) for _ in range(4)]
[t.start() for t in threads]
[q.put(x) for x in items]
q.join()                                 # wait until everything is done
[t.join(timeout=1) for t in threads]

# --- Shared state and races ----------------------------------------------
counter = 0
lock = threading.Lock()

def bump():
    global counter
    with lock:                           # critical section
        counter += 1                 # non-atomic without the lock!

# The GIL does NOT make this safe: += is read-modify-write (multiple bytecodes)
# Rule: protect any shared mutable state; or avoid sharing entirely (preferred).
from threading import RLock, Semaphore, Event, Condition
sem = Semaphore(5)                       # limit concurrency to 5
evt = Event()                            # one thread signals others
local = threading.local()                # per-thread storage — the escape hatch from sharing

# Timeouts and deadlines
if not evt.wait(timeout=5):
    raise TimeoutError("worker never signalled")

# Stop being scared of threads: the safe patterns are
#  1. share nothing (pass data in/out of submit)   2. queue.Queue for handoff
#  3. lock only tiny critical sections              4. never hold a lock across I/O
#  5. use a pool instead of one thread per task

15.3 Processes (CPU-bound) and what free-threading changes

from concurrent.futures import ProcessPoolExecutor
import multiprocessing as mp

def crunch(n: int) -> int:                      # MUST be importable (picklable)
    return sum(i * i for i in range(n))

if __name__ == "__main__":                      # required on Windows/macOS spawn
    with ProcessPoolExecutor(max_workers=mp.cpu_count()) as pool:
        results = list(pool.map(crunch, range(8), chunksize=1))
        print(results)

    # ProcessPoolExecutor is the friendly API; mp.Pool is the classic one
    with mp.Pool() as p:
        print(p.starmap(sorted, [(xs,), (ys,)]))

    # Imap for streaming results in order
    for r in pool.map(crunch, big_list):

Process gotchas: arguments and results are pickled (so closures, local functions, open sockets and file handles fail); __main__ must be import-safe (no side effects at import); start method differs by OS — fork on Linux (fast, but thread-unsafe), spawn on Windows/macOS (slow, safe). Set it explicitly: mp.set_start_method("spawn", force=True). For shared read-mostly data use mp.shared_memory or pass a memory-mapped NumPy array instead of copying.

15.4 asyncio — cooperative concurrency

import asyncio, httpx, time

async def fetch(client: httpx.AsyncClient, url: str) -> int:
    r = await client.get(url)                 # yields control while waiting
    return len(r.content)

async def main() -> None:
    urls = [f"https://httpbin.org/delay/1?i={i}" for i in range(20)]

    # 1 · Sequentially (slow): each await waits for the previous
    async with httpx.AsyncClient(timeout=10) as client:
        start = time.perf_counter()
        seq = [await fetch(client, u) for u in urls]         # 20 × ~1s ≈ 20s
        print("sequential", time.perf_counter() - start)

    # 2 · Concurrently with a TaskGroup (3.11+, structured concurrency — PREFER THIS)
    async with httpx.AsyncClient(timeout=10) as client:
        start = time.perf_counter()
        async with asyncio.TaskGroup() as tg:
            tasks = [tg.create_task(fetch(client, u)) for u in urls]
        sizes = [t.result() for t in tasks]                   # ≈1s total
        print("taskgroup", time.perf_counter() - start)

    # 3 · gather (classic API; return_exceptions=True to collect errors instead of raising)
    results = await asyncio.gather(*(fetch(client, u) for u in urls), return_exceptions=True)

    # 4 · Bound the concurrency with a semaphore
    sem = asyncio.Semaphore(10)
    async def limited(u):
        async with sem:
            return await fetch(client, u)
    await asyncio.gather(*(limited(u) for u in urls))

    # 5 · Stream results as they complete
    for coro in asyncio.as_completed([fetch(client, u) for u in urls]):
        print(await coro)

    # 6 · Timeouts and cancellation
    try:
        async with asyncio.timeout(5):        # 3.11+
            await fetch(client, urls[0])
    except TimeoutError:
        print("too slow")

    # 7 · Async generators + async for
    async def pages(client):
        for i in range(3):
            yield await fetch(client, urls[i])
    async for size in pages(client):
        print(size)

    # 8 · Queues for producer/consumer
    q: asyncio.Queue[int] = asyncio.Queue(maxsize=10)
    async def producer():
        for i in range(10):
            await q.put(i); await asyncio.sleep(0.01)
    async def consumer():
        while True:
            item = await q.get(); q.task_done()
    async with asyncio.TaskGroup() as tg:
        tg.create_task(producer()); tg.create_task(consumer())

    # 9 · Offload blocking work so it does not freeze the loop
    data = await asyncio.to_thread(blocking_db_call)          # thread
    # heavy CPU: use a process pool instead
    loop = asyncio.get_running_loop()
    with ProcessPoolExecutor() as pool:
        n = await loop.run_in_executor(pool, crunch, 10_000_000)

asyncio.run(main())          # the ONLY entry point you should use
Async red flags to look for in code review: time.sleep() inside async def (use asyncio.sleep); requests.get() inside async def (use httpx/aiohttp); subprocess.run() or a big for loop of CPU work inside a coroutine; a bare create_task() whose return value is thrown away (task may vanish); calling asyncio.run() inside a running loop (use await instead).

15.5 Choosing — and combining

ScenarioSolutionWhy
Fetch 500 URLsasyncio + httpxThousands of light tasks, tiny memory per task.
Fetch 10 URLs in a script that already uses requestsThreadPoolExecutor(10)Three lines, no rewrite.
Resize 20 000 imagesProcessPoolExecutor (or Pillow’s own threads)CPU + C-library work; processes bypass the GIL.
Web server handling 10 000 connectionsAsync framework (FastAPI/Starlette, aiohttp)Concurrency per connection without threads.
Numeric pipeline on 100 M rowsNumPy / polars / DuckDB (single process, vectorised)Better than naive parallelism; C-level loops release the GIL.
ML trainingGPU (PyTorch) + DataLoader workersProcesses for input, device for compute.
Long background jobsQueue + workers (Celery, RQ, arq, Dramatiq)Survives restarts, scales horizontally, retries built in.
Big data across many machinesRay / Dask / SparkDistributed scheduling and shared memory.
Measure before parallelising. Profile single-threaded first (ch. 17). Many “slow” programs are I/O-waiting on 5 calls that could be batched, or doing O(n²) work that a dict/set fixes. Concurrency adds real complexity: races, deadlocks, harder debugging, worse tracebacks.
