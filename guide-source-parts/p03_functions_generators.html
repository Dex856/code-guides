Arguments, closures, the LEGB rule, first-class functions, and decorators explained with a picture.

6.1 Defining and calling

def greet(name: str, punct: str = "!", *, upper: bool = False) -> str:
    """One-line summary.

    Args:
        name: who to greet.
        punct: trailing punctuation.
        upper: shout it.
    """
    msg = f"hi {name}{punct}"
    return msg.upper() if upper else msg

greet("Ada")                      # positional
greet("Ada", "!!")                # positional
greet(name="Ada", punct="?")      # keyword
greet("Ada", upper=True)          # keyword-only (after the bare *)

def area(w, /, h, *, unit="cm"):  # `/` = positional-only, `*` = keyword-only
    return w * h

# ⚠ THE #1 PYTHON TRAP: mutable default arguments
def bad(item, bucket=[]):        # the SAME list is reused for every call
    bucket.append(item); return bucket
bad(1); bad(2)                   # → [1] then [1, 2]  (!)

def good(item, bucket=None):     # fix: sentinel default, create inside
    if bucket is None:
        bucket = []
    bucket.append(item); return bucket
Why it happens: default values are evaluated once, when the def statement runs, and stored on the function object (f.__defaults__). Applies to [] {} set() and any custom mutable object. Use None, or better, a sentinel: _MISSING = object() — needed when None is a meaningful value.

6.2 Scope: the LEGB rule

count = 0
def bump():
    global count        # without this → UnboundLocalError
    count += 1

def outer():
    n = 0
    def inner():
        nonlocal n      # rebinds outer's n
        n += 1
        return n
    return inner

# Late binding: closures look up the variable when CALLED
funcs = [lambda: i for i in range(3)]
[f() for f in funcs]                  # [2, 2, 2]  ← surprise!
funcs = [lambda i=i: i for i in range(3)]     # bind now via default arg
[f() for f in funcs]                  # [0, 1, 2]
# or use functools.partial / a factory function
from functools import partial
funcs = [partial(lambda i: i, i) for i in range(3)]
# Python 3.12+ PEP 709 inlines comprehensions, so the classic
#   [lambda: x for x in range(3)] gotcha no longer leaks `x` — but the
#   late-binding rule for lambdas in loops still stands.

# Inspecting scope at runtime
def probe():
    x = 1

    def run(x):

        return x

6.4 Decorators — with and without the magic

A decorator is just a function that takes a function and returns a replacement. @timer above def f is exactly f = timer(f).

from functools import wraps
import time, logging

def timer(func):
    @wraps(func)                      # preserves __name__, __doc__, signature
    def wrapper(*args, **kwargs):
        t0 = time.perf_counter()
        try:
            return func(*args, **kwargs)
        finally:
            dt = (time.perf_counter() - t0) * 1000
            print(f"{func.__name__} took {dt:.2f} ms")
    return wrapper

def retry(times=3, delay=0.5, exceptions=(Exception,)):     # decorator WITH arguments
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(1, times + 1):
                try:
                    return func(*args, **kwargs)
                except exceptions:
                    if attempt == times:
                        raise
                    time.sleep(delay * attempt)
        return wrapper
    return decorator

@timer
@retry(times=3, exceptions=(TimeoutError,))
def fetch(url: str) -> str: ...

7 Iterators, generators and lazy pipelines

Process 100 GB of data in constant memory, and understand why your iterator “ran out”.

7.1 Generators: functions that remember where they stopped

def countdown(n):
    print("start")                 # runs only on first next()
    while n > 0:
        yield n                    # hand out a value, then PAUSE here
        n -= 1
    # falling off the end raises StopIteration automatically

g = countdown(3)
next(g)     # 'start' printed, returns 3
next(g)     # 2
list(g)     # [1]  (the generator is now exhausted)

# Streaming a huge file — constant memory regardless of file size
def read_records(path):
    with open(path, encoding="utf-8") as fh:
        header = next(fh).rstrip("\n").split(",")
        for line in fh:
            yield dict(zip(header, line.rstrip("\n").split(",")))

for rec in read_records("huge.csv"):
    process(rec)                   # never loads the whole file

# Infinite generator + islice = the safe pattern
import itertools
def naturals():
    n = 0
    while True:
        yield n
        n += 1
first_ten = list(itertools.islice(naturals(), 10))

# yield from: delegate to a sub-iterator (flattens easily)
def chain_all(*iterables):
    for it in iterables:
        yield from it
list(chain_all([1,2], "ab", range(2)))     # [1, 2, 'a', 'b', 0, 1]

# Two-way generators: send / throw / close
def accumulator():
    total = 0
    while True:
        x = yield total            # yield returns what was sent in
        if x is None:
            break
        total += x

acc = accumulator(); next(acc)     # prime it
acc.send(10), acc.send(5)          # 10, 15
acc.close()                        # GeneratorExit at the yield point

# @contextmanager is a generator in disguise (see ch. 10)
from contextlib import contextmanager
@contextmanager
def timed(label):
    t0 = time.perf_counter()
    yield                                   # control passes to the with-block here
    print(f"{label}: {time.perf_counter() - t0:.3f}s")

with timed("query"):
    run_query()

7.2 Lazy pipelines — the mental model

# The same task, four ways (slowest → fastest/most memory-efficient)
total = 0
for line in open(path):
    if "ERROR" in line:
        total += 1                                    # explicit loop: fine, verbose

total = sum(1 for line in open(path) if "ERROR" in line)     # generator: best default

total = len([l for l in open(path) if "ERROR" in l])         # list: wastes memory

# Chained lazy pipeline
from itertools import islice
pipeline = (
    line.rstrip("\n").split("|")
    for line in open(path)
)
errors = (row for row in pipeline if row[2] == "ERROR")
for row in islice(errors, 100):
    print(row)

# Generator expressions are not just "lazy comprehensions":
# they can feed any consumer that accepts an iterable
any(x > 100 for x in stream)          # short-circuits: stops at the first hit
min((f(x), x) for x in stream)        # streaming minimum with key
sum(len(w) for w in words) / len(words)
Choose the consumer, not the producer: sum, any, all, min, max, join, dict(), set(), deque(maxlen=…) and itertools all consume lazily. Keeping data in generator-land as long as possible is the cheapest performance optimisation there is.

# Sliding-window average without pandas
def moving_average(xs, window=3):
    from collections import deque
    from itertools import islice
    it = iter(xs)
    buf = deque(islice(it, window), maxlen=window)
    yield sum(buf) / len(buf)
    for x in it:
        buf.append(x)
        yield sum(buf) / len(buf)
Remember: groupby groups consecutive items — sort first. And every itertools function returns an iterator: materialise with list() only when you need to index or reuse it.
