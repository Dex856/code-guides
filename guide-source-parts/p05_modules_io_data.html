10 Modules, packages and imports

How the import system really finds your code — and how to structure a project so it never gets confused.

10.1 The import machine

# Import syntax you will actually use
import json                          # module object: json.loads(...)
import numpy as np                   # alias — community convention (np, pd, plt, tf)
from pathlib import Path             # names into your namespace
from collections import deque, Counter
from . import sibling                # relative import INSIDE a package
from ..models import User            # parent package
from mypkg.utils import helper as h

# Never do this (pollutes namespace, hides errors, breaks tooling)
from module import *

# Lazy import: only pay the cost when the branch is taken
def to_pdf(html: str) -> bytes:
    import weasyprint                # heavy optional dependency
    return weasyprint.HTML(string=html).write_pdf()

# Reload during development (not for production code)
import importlib, mymod
importlib.reload(mymod)

# Introspection
import sys
sys.modules["json"]                  # the cached module object
mymod.__file__                       # where it came from
mymod.__name__, mymod.__package__
dir(mymod)                           # public names
help(mymod.func)                     # docstring + signature
Modules are executed once, then cached. Importing pkg.a in five files runs pkg/a.py once total. Consequences: module-level mutable state is shared process-wide (great for caches, dangerous for “per-request” data), and a circular import only fails if the names you need are not yet defined when the cycle closes — the fix is to import inside a function or restructure into a lower-level module.

Circular imports — the three fixes

# Problem: a.py imports b, b.py imports a → ImportError (partially initialised module)
# Fix 1 (best): extract shared pieces into a third, lower-level module
#        a.py -> common.py ; b.py -> common.py
# Fix 2: import the MODULE, not the name, and use it lazily
import b            # then call b.thing() inside the function, not at import time
# Fix 3: import inside the function that needs it
def run():
    from . import heavy
    return heavy.work()

# Detect them early:
$ python -X importtime -c "import my_project"     # prints a timing tree of imports
$ uv run python -X importtime app.py 2> imports.txt
# Goal: keep import time small (fast CLI startup, quick tests, snappy serverless).

11 Files, paths, and data formats

Read/write files safely, navigate the filesystem with pathlib, and handle CSV, JSON, TOML, regex, dates, HTTP and SQLite.

11.1 Files: always with with, always with an encoding

from pathlib import Path
p = Path("data/report.csv")

# Text mode — specify encoding explicitly (don't rely on the locale)
with p.open("r", encoding="utf-8") as fh:
    text = fh.read()                       # whole file into memory
    fh.seek(0)
    lines = fh.readlines()                 # list of lines (keeps \n)
    fh.seek(0)
    for line in fh:                        # ← memory-efficient: one line at a time
        line = line.rstrip("\n")

# Writing (overwrite) and appending
Path("out.txt").write_text("hello\n", encoding="utf-8")
with Path("out.txt").open("a", encoding="utf-8") as fh:
    fh.write("more\n")

# Binary mode — no encoding, bytes in / bytes out
with Path("photo.jpg").open("rb") as fh:
    head = fh.read(16)                     # sniff magic bytes
with Path("copy.jpg").open("wb") as fh:
    fh.write(head)
bytes_data = Path("photo.jpg").read_bytes()

# Atomic writes: write to a temp file, then replace — never leave half a file
import os, tempfile
def atomic_write(path: Path, data: str) -> None:
    tmp = tempfile.NamedTemporaryFile("w", dir=path.parent, delete=False, encoding="utf-8")
    try:
        tmp.write(data); tmp.flush(); os.fsync(tmp.fileno())
    finally:
        tmp.close()
    os.replace(tmp.name, path)             # atomic on POSIX and Windows

# Reading very large files without loading them
def iter_chunks(path: Path, size: int = 1 << 20) -> Iterator[bytes]:
    with path.open("rb") as fh:
        while chunk := fh.read(size):
            yield chunk

# File modes cheat sheet:
# r read  ·  w write(truncate)  ·  a append  ·  x exclusive-create  ·  + read&write
# add b for binary (rb, wb), t is the default text mode
Encoding bugs are the most common file bug. On Windows the default is often a legacy codepage, not UTF-8 — always pass encoding="utf-8". Python 3.15 makes UTF-8 the default for text files (PEP 686), but explicit is still better. For files with a BOM use encoding="utf-8-sig"; for files from Excel/Windows, cp1252 or latin-1 may be needed. On decode errors: errors="replace" or errors="ignore" when you must not crash.

import re
pattern = re.compile(r"\b\d{1,3}(?:\.\d{1,3}){3}\b")     # IPv4-ish, compiled once

re.search(r"\d+", "abc 123")            # <Match '123'>
re.findall(r"\w+@\w+\.\w+", text)       # ['a@b.com', ...]
re.sub(r"\s+", " ", text)               # collapse whitespace
re.sub(r"(\w+)@(\w+)", r"\2:\1", "me@here")      # backrefs in the replacement
re.split(r"\s*,\s*", "a, b ,c")         # ['a','b','c']
[m.groupdict() for m in re.finditer(pattern, log_text)]
re.sub(r"(?i)password=\S+", "password=***", log_line)       # redact secrets

# Named groups + verbose mode = maintainable regexes
DATE = re.compile(r"""
    (?P<year>\d{4}) - (?P<month>\d{2}) - (?P<day>\d{2})
""", re.VERBOSE)
m = DATE.search("on 2026-10-02")
m.groupdict()

# ⚠ Catastrophic backtracking: nested quantifiers on ambiguous patterns
# BAD:  ^(\w+\s?)*$      GOOD: ^\w+(?:\s\w+)*$
# Mitigation: make the pattern specific, avoid nested quantifiers, or use the
# `regex` package which supports timeouts / atomic groups.

text = "Contact: ada@example.com or linus@kernel.org"

# --- SQLite: a real database in one file ---------------------------------
import sqlite3
con = sqlite3.connect("app.db")
con.row_factory = sqlite3.Row                    # rows behave like dicts
con.execute("PRAGMA journal_mode=WAL")           # better concurrency

con.execute("""CREATE TABLE IF NOT EXISTS orders(
    id INTEGER PRIMARY KEY, sku TEXT NOT NULL, qty INTEGER, created TEXT DEFAULT CURRENT_TIMESTAMP)""")

# ALWAYS parameterise — never f-string values into SQL (injection)
con.execute("INSERT INTO orders(sku, qty) VALUES (?, ?)", ("A1", 3))
con.executemany("INSERT INTO orders(sku, qty) VALUES (?, ?)", [("B2", 1), ("C3", 5)])
con.commit()

for row in con.execute("SELECT sku, qty FROM orders WHERE qty > ? ORDER BY qty DESC", (1,)):
    print(row["sku"], row["qty"])

with con:                                        # transaction: commit on success, rollback on error
    con.execute("UPDATE orders SET qty = qty + 1 WHERE sku = ?", ("A1",))

con.execute("SELECT count(*) FROM orders").fetchone()[0]
con.close()
